import React, { useState, useEffect } from 'react';
import { 
  Map, 
  Search, 
  AlertTriangle, 
  Navigation, 
  MapPin, 
  Clock, 
  Home, 
  Briefcase, 
  ChevronLeft, 
  CornerUpLeft, 
  X, 
  ShieldAlert, 
  Car, 
  Waves, 
  TrafficCone, 
  Flag,
  Mic,
  Camera,
  Map as MapIcon
} from 'lucide-react';

const WazePakistanApp = () => {
  const [currentScreen, setCurrentScreen] = useState('map');
  const [destination, setDestination] = useState('');
  const [toastMessage, setToastMessage] = useState(null);

  const navigateTo = (screen) => setCurrentScreen(screen);

  const showToast = (message) => {
    setToastMessage(message);
    setTimeout(() => setToastMessage(null), 3000);
  };

  const handleStartNavigation = (dest) => {
    setDestination(dest);
    setCurrentScreen('navigation');
  };

  const handleReport = (type) => {
    showToast(`${type} reported successfully! Thank you.`);
    setCurrentScreen('map');
  };

  // --- Reusable Components ---

  const TopNotch = () => (
    <div className="absolute top-0 inset-x-0 h-6 flex justify-center z-50">
      <div className="w-40 h-6 bg-black rounded-b-3xl"></div>
    </div>
  );

  const StatusBar = () => (
    <div className="absolute top-0 inset-x-0 h-12 flex justify-between items-center px-6 pt-2 z-40 text-xs font-bold text-slate-800">
      <span>12:45</span>
      <div className="flex space-x-2">
        <span>LTE</span>
        <div className="w-5 h-3 bg-slate-800 rounded-sm"></div>
      </div>
    </div>
  );

  const BottomTabBar = () => (
    <div className="absolute bottom-0 w-full h-20 bg-white border-t border-gray-200 flex justify-around items-center px-2 pb-4 shadow-[0_-4px_15px_rgba(0,0,0,0.05)] z-40 rounded-b-[1.5rem]">
      <TabButton icon={<MapIcon size={24} />} label="Map" active={currentScreen === 'map'} onClick={() => navigateTo('map')} />
      <TabButton icon={<Search size={24} />} label="Search" active={currentScreen === 'search'} onClick={() => navigateTo('search')} />
      <TabButton icon={<AlertTriangle size={24} />} label="Report" active={currentScreen === 'report'} onClick={() => navigateTo('report')} />
      <TabButton icon={<Navigation size={24} />} label="Drive" active={currentScreen === 'navigation'} onClick={() => currentScreen !== 'navigation' ? navigateTo('navigation') : null} disabled={!destination && currentScreen !== 'navigation'} />
    </div>
  );

  const TabButton = ({ icon, label, active, onClick, disabled }) => (
    <button 
      onClick={onClick}
      disabled={disabled}
      className={`flex flex-col items-center justify-center w-16 h-12 rounded-xl transition-all duration-200 ${
        active ? 'text-blue-600 scale-110' : 'text-gray-400 hover:text-gray-600'
      } ${disabled ? 'opacity-30 cursor-not-allowed' : ''}`}
    >
      {icon}
      <span className="text-[10px] font-bold mt-1">{label}</span>
    </button>
  );

  const MapBackground = ({ children, isNavigating }) => (
    <div className="absolute inset-0 bg-[#e5f0ea] overflow-hidden">
      {/* Grid pattern to simulate map blocks */}
      <div className="absolute inset-0 opacity-20" style={{ backgroundImage: 'radial-gradient(#475569 1px, transparent 1px)', backgroundSize: '20px 20px' }}></div>
      
      {/* Decorative Roads */}
      <div className={`absolute w-full h-12 bg-white ${isNavigating ? 'rotate-[-20deg] top-1/2 scale-150' : 'top-1/3'} left-0 shadow-sm transition-all duration-700`}></div>
      <div className={`absolute w-12 h-full bg-white ${isNavigating ? 'rotate-[15deg] left-1/2 scale-150' : 'left-1/3'} top-0 shadow-sm transition-all duration-700`}></div>
      <div className={`absolute w-full h-8 bg-white ${isNavigating ? 'hidden' : 'bottom-1/4'} left-0 shadow-sm`}></div>
      <div className={`absolute w-8 h-full bg-white ${isNavigating ? 'hidden' : 'right-1/4'} top-0 shadow-sm`}></div>
      
      {/* Navigating Route Line */}
      {isNavigating && (
        <div className="absolute w-3 h-64 bg-purple-500 left-1/2 top-1/4 rotate-[15deg] transform -translate-x-1/2 rounded-full shadow-[0_0_10px_rgba(168,85,247,0.5)] z-10"></div>
      )}

      {children}
    </div>
  );

  // --- Screens ---

  const MapScreen = () => (
    <div className="relative w-full h-full flex flex-col bg-transparent">
      <MapBackground isNavigating={false}>
        {/* Map Markers */}
        <div className="absolute top-1/3 left-1/3 w-8 h-8 bg-white rounded-full shadow-lg flex items-center justify-center transform -translate-x-4 -translate-y-4 border-2 border-gray-100 z-10">
          <Car size={16} className="text-blue-500" />
        </div>
        <div className="absolute bottom-1/3 right-1/4 w-8 h-8 bg-white rounded-full shadow-lg flex items-center justify-center transform -translate-x-4 -translate-y-4 border-2 border-gray-100 z-10">
          <ShieldAlert size={16} className="text-amber-500" />
        </div>
      </MapBackground>

      {/* Top Search Bar */}
      <div className="absolute top-14 inset-x-4 z-20">
        <div 
          onClick={() => navigateTo('search')}
          className="bg-white rounded-full shadow-lg h-14 flex items-center px-4 cursor-pointer hover:bg-gray-50 transition-colors border border-gray-100"
        >
          <Search className="text-gray-400 mr-3" size={20} />
          <span className="text-gray-500 font-medium flex-1">Where to?</span>
          <div className="w-8 h-8 bg-gray-100 rounded-full flex items-center justify-center">
            <Mic className="text-gray-600" size={16} />
          </div>
        </div>
      </div>

      {/* Floating Action Buttons */}
      <div className="absolute bottom-24 right-4 flex flex-col space-y-4 z-20">
        <button 
          className="w-14 h-14 bg-white rounded-full shadow-lg flex items-center justify-center text-gray-700 font-bold border border-gray-100"
        >
          <MapIcon size={24} className="text-blue-500" />
        </button>
        <button 
          onClick={() => navigateTo('report')}
          className="w-16 h-16 bg-amber-400 rounded-full shadow-xl flex items-center justify-center text-white border-2 border-white transform hover:scale-105 active:scale-95 transition-all"
        >
          <AlertTriangle size={32} fill="white" className="text-amber-500" />
        </button>
      </div>

      {/* Speedometer */}
      <div className="absolute bottom-24 left-4 w-16 h-16 bg-white rounded-full shadow-lg border-4 border-gray-100 flex flex-col items-center justify-center z-20">
        <span className="text-xl font-black text-slate-800 leading-none">60</span>
        <span className="text-[10px] text-gray-500 font-bold uppercase">km/h</span>
      </div>
    </div>
  );

  const SearchScreen = () => (
    <div className="relative w-full h-full bg-slate-50 flex flex-col z-30 pb-20 overflow-y-auto">
      <div className="bg-white px-4 pt-14 pb-4 shadow-sm relative">
        <div className="flex items-center">
          <button onClick={() => navigateTo('map')} className="p-2 -ml-2 text-gray-600 rounded-full hover:bg-gray-100">
            <ChevronLeft size={28} />
          </button>
          <div className="flex-1 ml-2 relative">
            <input 
              autoFocus
              type="text" 
              placeholder="Where to?" 
              className="w-full bg-gray-100 rounded-full h-12 pl-4 pr-10 outline-none text-slate-800 font-medium focus:ring-2 ring-blue-400"
            />
            <Search className="absolute right-3 top-3 text-gray-400" size={20} />
          </div>
        </div>
      </div>

      <div className="p-4 space-y-6">
        {/* Favorites */}
        <div>
          <h3 className="text-sm font-bold text-gray-400 uppercase mb-3 px-2">Favorites</h3>
          <div className="flex space-x-4 px-2">
            <button 
              onClick={() => handleStartNavigation('Home')}
              className="flex-1 bg-white p-4 rounded-2xl shadow-sm flex flex-col items-center justify-center space-y-2 hover:bg-blue-50 transition-colors border border-gray-100"
            >
              <div className="w-10 h-10 bg-blue-100 rounded-full flex items-center justify-center">
                <Home className="text-blue-500" size={20} />
              </div>
              <span className="font-bold text-slate-700">Home</span>
            </button>
            <button 
              onClick={() => handleStartNavigation('Work')}
              className="flex-1 bg-white p-4 rounded-2xl shadow-sm flex flex-col items-center justify-center space-y-2 hover:bg-blue-50 transition-colors border border-gray-100"
            >
              <div className="w-10 h-10 bg-blue-100 rounded-full flex items-center justify-center">
                <Briefcase className="text-blue-500" size={20} />
              </div>
              <span className="font-bold text-slate-700">Work</span>
            </button>
          </div>
        </div>

        {/* Recents */}
        <div>
          <h3 className="text-sm font-bold text-gray-400 uppercase mb-2 px-2">Recent</h3>
          <div className="bg-white rounded-3xl shadow-sm border border-gray-100 overflow-hidden">
            {[
              { name: 'Lahore Fort', address: 'Fort Road, Walled City of Lahore', icon: <MapPin /> },
              { name: 'Faisal Mosque', address: 'Shah Faisal Ave, E-8, Islamabad', icon: <Clock /> },
              { name: 'Jinnah International Airport', address: 'Airport Road, Karachi', icon: <Clock /> },
            ].map((item, idx) => (
              <button 
                key={idx}
                onClick={() => handleStartNavigation(item.name)}
                className={`w-full flex items-center p-4 text-left hover:bg-gray-50 transition-colors ${idx !== 2 ? 'border-b border-gray-100' : ''}`}
              >
                <div className="w-8 h-8 text-gray-400 mr-4 flex-shrink-0">
                  {item.icon}
                </div>
                <div className="flex-1 min-w-0">
                  <p className="font-bold text-slate-800 truncate">{item.name}</p>
                  <p className="text-sm text-gray-500 truncate">{item.address}</p>
                </div>
              </button>
            ))}
          </div>
        </div>
      </div>
    </div>
  );

  const NavigationScreen = () => (
    <div className="relative w-full h-full flex flex-col bg-transparent">
      <MapBackground isNavigating={true}>
        {/* User Car Icon */}
        <div className="absolute top-1/2 left-1/2 w-12 h-12 bg-white rounded-full shadow-2xl flex items-center justify-center transform -translate-x-1/2 -translate-y-1/2 border-[3px] border-blue-500 z-20">
          <Car size={24} className="text-blue-600" />
          {/* Headlights effect */}
          <div className="absolute -top-6 left-1/2 transform -translate-x-1/2 w-16 h-12 bg-gradient-to-t from-yellow-200 to-transparent opacity-50 blur-sm pointer-events-none rounded-t-full"></div>
        </div>
      </MapBackground>

      {/* Top Instruction Banner */}
      <div className="absolute top-12 inset-x-2 z-30 flex flex-col">
        <div className="bg-green-500 rounded-2xl shadow-lg text-white p-4 flex items-center border border-green-600">
          <div className="mr-4 text-white">
            <CornerUpLeft size={36} strokeWidth={3} />
          </div>
          <div className="flex-1">
            <div className="text-xl font-black leading-tight">200 m</div>
            <div className="text-sm font-medium opacity-90 truncate">Turn left onto Shahrah-e-Faisal</div>
          </div>
        </div>
      </div>

      {/* Bottom ETA Panel (covers tab bar) */}
      <div className="absolute bottom-0 w-full bg-white rounded-t-3xl shadow-[0_-10px_25px_rgba(0,0,0,0.1)] z-50 p-6">
        <div className="flex justify-between items-end mb-4">
          <div>
            <div className="text-3xl font-black text-slate-800">25 min</div>
            <div className="text-gray-500 font-medium text-sm mt-1">14 km • 12:45 PM arrival</div>
          </div>
          <button 
            onClick={() => navigateTo('map')}
            className="bg-red-100 text-red-600 px-6 py-3 rounded-full font-bold shadow-sm hover:bg-red-200 transition-colors"
          >
            Stop
          </button>
        </div>
        <div className="w-full bg-gray-200 rounded-full h-2 mb-2">
          <div className="bg-purple-500 h-2 rounded-full w-1/3"></div>
        </div>
        <div className="text-center text-xs text-gray-400 font-bold uppercase mt-2">
          Navigating to {destination || 'Destination'}
        </div>
      </div>
      
      {/* Speedometer during Nav */}
      <div className="absolute bottom-40 left-4 w-16 h-16 bg-white rounded-full shadow-lg border-4 border-red-500 flex flex-col items-center justify-center z-20">
        <span className="text-xl font-black text-red-600 leading-none">85</span>
        <span className="text-[10px] text-gray-500 font-bold uppercase">km/h</span>
      </div>
    </div>
  );

  const ReportScreen = () => {
    const reportOptions = [
      { id: 'police', label: 'Police Naka', icon: <ShieldAlert size={32} />, color: 'bg-blue-500' },
      { id: 'traffic', label: 'Heavy Traffic', icon: <Car size={32} />, color: 'bg-red-500' },
      { id: 'block', label: 'Procession', icon: <Flag size={32} />, color: 'bg-orange-500' },
      { id: 'pothole', label: 'Deep Pothole', icon: <AlertTriangle size={32} />, color: 'bg-amber-500' },
      { id: 'water', label: 'Waterlogging', icon: <Waves size={32} />, color: 'bg-cyan-500' },
      { id: 'hazard', label: 'Hazard', icon: <TrafficCone size={32} />, color: 'bg-yellow-500' },
      { id: 'camera', label: 'Camera', icon: <Camera size={32} />, color: 'bg-slate-700' },
      { id: 'closure', label: 'Closure', icon: <X size={32} />, color: 'bg-red-600' },
    ];

    return (
      <div className="absolute inset-0 bg-white z-50 flex flex-col rounded-[2rem] overflow-hidden">
        <div className="bg-amber-400 text-white pt-14 pb-6 px-4 shadow-md relative rounded-b-3xl z-10">
          <button 
            onClick={() => navigateTo('map')}
            className="absolute top-14 left-4 p-2 bg-white/20 rounded-full hover:bg-white/30 transition-colors"
          >
            <X size={24} />
          </button>
          <h2 className="text-2xl font-black text-center mt-2">What do you see?</h2>
        </div>
        
        <div className="flex-1 p-6 overflow-y-auto pb-24">
          <div className="grid grid-cols-3 gap-4">
            {reportOptions.map((opt) => (
              <button 
                key={opt.id}
                onClick={() => handleReport(opt.label)}
                className="flex flex-col items-center justify-center space-y-3"
              >
                <div className={`w-20 h-20 rounded-full shadow-lg ${opt.color} flex items-center justify-center text-white border-[3px] border-white transform hover:scale-105 active:scale-95 transition-all`}>
                  {opt.icon}
                </div>
                <span className="text-xs font-bold text-slate-700 text-center leading-tight max-w-[80px]">
                  {opt.label}
                </span>
              </button>
            ))}
          </div>
        </div>
      </div>
    );
  };

  return (
    <div className="flex items-center justify-center min-h-screen bg-gray-900 font-sans selection:bg-blue-200 p-4">
      {/* Phone Container */}
      <div className="w-full max-w-[400px] h-[800px] max-h-[90vh] bg-white relative overflow-hidden shadow-2xl rounded-[2.5rem] border-[12px] border-slate-800 flex flex-col ring-4 ring-gray-900/50">
        
        <StatusBar />
        <TopNotch />

        {/* Dynamic Screen Rendering */}
        <div className="flex-1 relative overflow-hidden">
          {currentScreen === 'map' && <MapScreen />}
          {currentScreen === 'search' && <SearchScreen />}
          {currentScreen === 'navigation' && <NavigationScreen />}
          {currentScreen === 'report' && <ReportScreen />}
        </div>

        {/* Global Toast Notification */}
        {toastMessage && (
          <div className="absolute top-24 inset-x-4 z-[100] animate-bounce">
            <div className="bg-slate-800 text-white px-4 py-3 rounded-xl shadow-2xl font-medium text-sm text-center">
              {toastMessage}
            </div>
          </div>
        )}

        {/* Bottom Tab Bar (Hidden in Nav and Report) */}
        {currentScreen !== 'navigation' && currentScreen !== 'report' && (
          <BottomTabBar />
        )}

        {/* Home Indicator Line (iOS style) */}
        <div className="absolute bottom-1 inset-x-0 flex justify-center z-50 pointer-events-none">
          <div className="w-32 h-1 bg-gray-300 rounded-full"></div>
        </div>
      </div>
    </div>
  );
};

export default WazePakistanApp;
